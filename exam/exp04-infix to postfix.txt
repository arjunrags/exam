//infix to postfix
#include<stdio.h>
#include<ctype.h>
#define MAX 20
int top=-1;
char s[MAX];

void push(char e) { if(top<MAX-1) s[++top]=e; }
char pop() { return (top==-1) ? '\0' : s[top--]; }

int priority(char c) {
    if(c=='(') return 1;
    if(c=='+'||c=='-') return 2;
    if(c=='*'||c=='/'||c=='%') return 3;
    if(c=='^') return 4;
    return 0;
}

int main() {
    char infix[MAX], postfix[MAX];
    int i, j=0;
    printf("Enter Infix: ");
    scanf("%s", infix);
    
    for(i=0; infix[i]; i++) {
        if(isalnum(infix[i])) postfix[j++]=infix[i];
        else if(infix[i]=='(') push(infix[i]);
        else if(infix[i]==')') {
            while(s[top]!='(') postfix[j++]=pop();
            top--;
        } else {
            while(top!=-1 && priority(s[top])>=priority(infix[i]))
                postfix[j++]=pop();
            push(infix[i]);
        }
    }
    while(top!=-1) postfix[j++]=pop();
    postfix[j]='\0';
    printf("Postfix: %s\n", postfix);
    return 0;
}
