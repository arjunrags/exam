//bfs and dfs
#include<stdio.h>
#include<stdlib.h>

int a[50][50], n, visited[50];
int q[20], f=-1, r=-1;

void bfs(int v) {
    int cur;
    visited[v] = 1;
    q[++r] = v;
    while(f != r) {
        cur = q[++f];
        for(int i=1; i<=n; i++) {
            if(a[cur][i] && !visited[i]) {
                q[++r] = i;
                visited[i] = 1;
                printf("%d ", i);
            }
        }
    }
}

void dfs(int v) {
    visited[v] = 1;
    for(int i=1; i<=n; i++) {
        if(a[v][i] && !visited[i]) {
            printf("%d ", i);
            dfs(i);
        }
    }
}

int main() {
    int ch, start;
    printf("Enter vertices: ");
    scanf("%d", &n);
    printf("Enter adjacency matrix:\n");
    for(int i=1; i<=n; i++)
        for(int j=1; j<=n; j++)
            scanf("%d", &a[i][j]);
    
    for(int i=1; i<=n; i++) visited[i] = 0;
    
    printf("\n1.BFS 2.DFS 3.Exit\nChoice: ");
    scanf("%d", &ch);
    printf("Enter start vertex: ");
    scanf("%d", &start);
    
    switch(ch) {
        case 1:
            printf("BFS from %d: ", start);
            bfs(start);
            for(int i=1; i<=n; i++)
                if(!visited[i]) printf("\nNot reachable: %d", i);
            break;
        case 2:
            printf("DFS from %d: ", start);
            dfs(start);
            break;
        case 3: exit(0);
    }
    return 0;

}
