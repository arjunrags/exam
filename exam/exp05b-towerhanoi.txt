//towerhanoi
#include<stdio.h>
void tower(int n, char s, char d, char t) {
    if(n == 1) {
        printf("\nMove disk %d from %c to %c", n, s, d);
        return;
    }
    tower(n-1, s, t, d);
    printf("\nMove disk %d from %c to %c", n, s, d);
    tower(n-1, t, d, s);
}
int main() {
    int n;
    printf("Enter number of disks: ");
    scanf("%d", &n);
    tower(n, 'A', 'B', 'C');
    return 0;
}
