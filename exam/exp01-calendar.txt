//calendar
#include<stdio.h>
#include<stdlib.h>

struct Day {
    char *name;
    int date;
    char *activity;
};

struct Day create() {
    struct Day d;
    d.name = (char*)malloc(20);
    d.activity = (char*)malloc(100);
    printf("Day Name: ");
    scanf("%s", d.name);
    printf("Date: ");
    scanf("%d", &d.date);
    printf("Activity: ");
    scanf(" %[^\n]", d.activity);
    return d;
}

void read(struct Day cal[], int n) {
    for(int i=0; i<n; i++) {
        printf("\nDay %d:\n", i+1);
        cal[i] = create();
    }
}

void display(struct Day cal[], int n) {
    printf("\nCalendar:\n");
    for(int i=0; i<n; i++) {
        printf("\n%d. %s (Date: %d)\n   %s\n", i+1, cal[i].name, cal[i].date, cal[i].activity);
    }
}

int main() {
    struct Day *cal = (struct Day*)malloc(7 * sizeof(struct Day));
    read(cal, 7);
    display(cal, 7);
    for(int i=0; i<7; i++) {
        free(cal[i].name);
        free(cal[i].activity);
    }
    free(cal);
    return 0;
}
