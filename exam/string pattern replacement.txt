//string pattern replacement
#include<stdio.h>
#include<string.h>
int main() {
    char str[100], pat[100], rep[100], ans[200];
    printf("Enter string: ");
    fgets(str, 100, stdin);
    str[strcspn(str, "\n")] = '\0';
    printf("Enter pattern: ");
    fgets(pat, 100, stdin);
    pat[strcspn(pat, "\n")] = '\0';
    printf("Enter replacement: ");
    fgets(rep, 100, stdin);
    rep[strcspn(rep, "\n")] = '\0';
    
    int i=0, j, m=0, flag=0;
    while(str[i]) {
        j=0;
        while(str[i+j] && pat[j] && str[i+j]==pat[j]) j++;
        if(!pat[j]) {
            flag=1;
            for(int k=0; rep[k]; k++) ans[m++]=rep[k];
            i+=j;
        } else ans[m++]=str[i++];
    }
    ans[m]='\0';
    printf(flag ? "Result: %s\n" : "Pattern not found\n", ans);
    return 0;
}
